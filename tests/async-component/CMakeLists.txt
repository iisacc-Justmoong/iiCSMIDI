cmake_minimum_required(VERSION 3.31)
project(iiCSMIDIComponentContract LANGUAGES CXX)
if(MIDI_REQUIRED)
    find_package(iiCSMIDI 0.3 CONFIG QUIET COMPONENTS Async)
    if(iiCSMIDI_FOUND OR NOT iiCSMIDI_NOT_FOUND_MESSAGE MATCHES "separately configured")
        message(FATAL_ERROR "Required legacy Async component must report its migration path")
    endif()
else()
    find_package(iiCSMIDI 0.3 CONFIG REQUIRED OPTIONAL_COMPONENTS Async)
    if(iiCSMIDI_Async_FOUND OR TARGET iiCSMIDI::Async)
        message(FATAL_ERROR "Optional legacy Async component must remain unavailable")
    endif()
endif()
