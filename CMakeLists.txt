cmake_minimum_required(VERSION 3.31)
project(iiCSMIDI VERSION 0.3.1 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
find_package(iiSDKCore 0.14 CONFIG REQUIRED COMPONENTS Contracts Runtime)
find_package(iiFileProviderNative 0.8 CONFIG REQUIRED)
find_package(iiJson 0.1 CONFIG REQUIRED)
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
set(_kind SHARED)
if(EMSCRIPTEN OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set(_kind STATIC)
endif()
add_library(iiCSMIDI ${_kind} src/iiCSMIDI.h src/iiCSMIDI.cpp src/Export.h src/MidiDocument.h src/MidiDocument.cpp)
add_library(iiCSMIDI::iiCSMIDI ALIAS iiCSMIDI)
target_compile_features(iiCSMIDI PUBLIC cxx_std_23)
target_compile_definitions(iiCSMIDI PRIVATE IICSMIDI_BUILDING_LIBRARY)
if(_kind STREQUAL "STATIC")
    target_compile_definitions(iiCSMIDI PUBLIC IICSMIDI_STATIC)
endif()
if(MSVC)
    target_compile_options(iiCSMIDI PUBLIC /utf-8 /Zc:__cplusplus)
endif()
target_link_libraries(iiCSMIDI PUBLIC iiSDKCore::Contracts iiFileProvider::Native PRIVATE iiJson::iiJson)
target_include_directories(iiCSMIDI PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(iiCSMIDI PROPERTIES CXX_EXTENSIONS OFF CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES VERSION ${PROJECT_VERSION} SOVERSION 0.3)
add_library(iiCSMIDIAsync INTERFACE)
add_library(iiCSMIDI::Async ALIAS iiCSMIDIAsync)
set_target_properties(iiCSMIDIAsync PROPERTIES EXPORT_NAME Async)
target_link_libraries(iiCSMIDIAsync INTERFACE iiCSMIDI::iiCSMIDI iiSDKCore::Runtime)
add_custom_target(iiCSMIDIBoundary ALL COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/boundary.py" "${CMAKE_CURRENT_SOURCE_DIR}" VERBATIM)
add_dependencies(iiCSMIDI iiCSMIDIBoundary)
if(BUILD_TESTING)
    foreach(_test hello_world authorship)
        add_executable(iiCSMIDI_${_test} tests/${_test}.cpp)
        target_link_libraries(iiCSMIDI_${_test} PRIVATE iiCSMIDI::iiCSMIDI)
        add_test(NAME iiCSMIDI.${_test} COMMAND iiCSMIDI_${_test} "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    endforeach()
    add_executable(iiCSMIDI_async tests/async.cpp)
    target_link_libraries(iiCSMIDI_async PRIVATE iiCSMIDI::Async)
    add_test(NAME iiCSMIDI.Async COMMAND iiCSMIDI_async)
    set_tests_properties(iiCSMIDI.Async PROPERTIES TIMEOUT 30)
    add_test(NAME iiCSMIDI.Boundary COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/boundary.py" "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
set(_package_dir "${CMAKE_INSTALL_LIBDIR}/cmake/iiCSMIDI")
configure_package_config_file(cmake/iiCSMIDIConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/iiCSMIDIConfig.cmake" INSTALL_DESTINATION "${_package_dir}")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/iiCSMIDIConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(TARGETS iiCSMIDI EXPORT iiCSMIDITargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(TARGETS iiCSMIDIAsync EXPORT iiCSMIDIAsyncTargets)
install(FILES src/iiCSMIDI.h src/Export.h src/MidiDocument.h src/MidiAsync.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT iiCSMIDITargets NAMESPACE iiCSMIDI:: DESTINATION "${_package_dir}")
install(EXPORT iiCSMIDIAsyncTargets NAMESPACE iiCSMIDI:: DESTINATION "${_package_dir}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/iiCSMIDIConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/iiCSMIDIConfigVersion.cmake" DESTINATION "${_package_dir}")
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/iiCSMIDI)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/iiCSMIDI/docs)
