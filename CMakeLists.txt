cmake_minimum_required(VERSION 3.31)
project(iiCSMIDI VERSION 0.3.1 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
option(BUILD_SHARED_LIBS "Build iiCSMIDI as a shared library" ON)
set(_kind SHARED)
if(NOT BUILD_SHARED_LIBS)
    set(_kind STATIC)
endif()
if(EMSCRIPTEN AND BUILD_SHARED_LIBS)
    set_property(GLOBAL PROPERTY TARGET_SUPPORTS_SHARED_LIBS TRUE)
endif()
add_library(iiCSMIDI ${_kind} src/iiCSMIDI.h src/iiCSMIDI.cpp src/Export.h src/MidiDocument.h src/MidiDocument.cpp)
add_library(iiCSMIDI::iiCSMIDI ALIAS iiCSMIDI)
target_compile_features(iiCSMIDI PUBLIC cxx_std_23)
target_compile_definitions(iiCSMIDI PRIVATE IICSMIDI_BUILDING_LIBRARY)
if(_kind STREQUAL "STATIC")
    target_compile_definitions(iiCSMIDI PUBLIC IICSMIDI_STATIC)
endif()
if(MSVC)
    target_compile_options(iiCSMIDI PUBLIC /utf-8 /Zc:__cplusplus)
endif()
target_include_directories(iiCSMIDI PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(iiCSMIDI PROPERTIES CXX_EXTENSIONS OFF CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES VERSION ${PROJECT_VERSION} SOVERSION 0.3)
if(BUILD_SHARED_LIBS)
    if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
        set_target_properties(iiCSMIDI PROPERTIES FRAMEWORK TRUE FRAMEWORK_VERSION A MACOSX_FRAMEWORK_IDENTIFIER "com.iisacc.iiCSMIDI")
    elseif(APPLE)
        set_property(TARGET iiCSMIDI APPEND PROPERTY INSTALL_RPATH "@loader_path")
    elseif(UNIX AND NOT EMSCRIPTEN)
        set_property(TARGET iiCSMIDI APPEND PROPERTY INSTALL_RPATH "$ORIGIN")
    endif()
    if(EMSCRIPTEN)
        target_compile_options(iiCSMIDI PRIVATE -fPIC -sSIDE_MODULE=1)
        target_link_options(iiCSMIDI PRIVATE -sSIDE_MODULE=1)
    endif()
endif()
add_custom_target(iiCSMIDIBoundary ALL COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/boundary.py" "${CMAKE_CURRENT_SOURCE_DIR}" VERBATIM)
add_dependencies(iiCSMIDI iiCSMIDIBoundary)
if(BUILD_TESTING)
    if(BUILD_SHARED_LIBS)
        add_test(NAME iiCSMIDI.shared_type COMMAND "${CMAKE_COMMAND}"
            "-DTARGET_TYPE=$<TARGET_PROPERTY:iiCSMIDI,TYPE>"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/shared_target_type.cmake")
    endif()
    foreach(_test hello_world independent)
        add_executable(iiCSMIDI_${_test} tests/${_test}.cpp)
        target_link_libraries(iiCSMIDI_${_test} PRIVATE iiCSMIDI::iiCSMIDI)
        add_test(NAME iiCSMIDI.${_test} COMMAND iiCSMIDI_${_test} "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
        set_tests_properties(iiCSMIDI.${_test} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/resources")
    endforeach()
    if(BUILD_SHARED_LIBS AND NOT CMAKE_CROSSCOMPILING)
        add_test(NAME iiCSMIDI.shared_link COMMAND "${CMAKE_COMMAND}"
            "-DIMAGE=$<TARGET_FILE:iiCSMIDI_independent>" "-DEXPECTED=$<TARGET_FILE:iiCSMIDI>"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/shared_link.cmake")
    endif()
    add_test(NAME iiCSMIDI.AsyncBoundary COMMAND "${CMAKE_COMMAND}"
        "-DMIDI_SOURCE=${CMAKE_CURRENT_SOURCE_DIR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/async_sdk_boundary.cmake")
    add_test(NAME iiCSMIDI.SDKBoundary COMMAND "${CMAKE_COMMAND}"
        "-DMIDI_SOURCE=${CMAKE_CURRENT_SOURCE_DIR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/sdk_independence.cmake")
    add_test(NAME iiCSMIDI.Boundary COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/boundary.py" "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
set(_package_dir "${CMAKE_INSTALL_LIBDIR}/cmake/iiCSMIDI")
configure_package_config_file(cmake/iiCSMIDIConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/iiCSMIDIConfig.cmake" INSTALL_DESTINATION "${_package_dir}")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/iiCSMIDIConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(TARGETS iiCSMIDI EXPORT iiCSMIDITargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} FRAMEWORK DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES src/iiCSMIDI.h src/Export.h src/MidiDocument.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT iiCSMIDITargets NAMESPACE iiCSMIDI:: DESTINATION "${_package_dir}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/iiCSMIDIConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/iiCSMIDIConfigVersion.cmake" DESTINATION "${_package_dir}")
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/iiCSMIDI)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/iiCSMIDI/docs
    PATTERN ".*.iisacc-preview" EXCLUDE PATTERN ".iisacc-temporary.lock" EXCLUDE PATTERN ".DS_Store" EXCLUDE)
install(DIRECTORY example/CoreAsync DESTINATION ${CMAKE_INSTALL_DATADIR}/iiCSMIDI/example
    PATTERN ".*.iisacc-preview" EXCLUDE PATTERN ".iisacc-temporary.lock" EXCLUDE PATTERN ".DS_Store" EXCLUDE)
install(DIRECTORY example/FileProvider DESTINATION ${CMAKE_INSTALL_DATADIR}/iiCSMIDI/example
    PATTERN ".*.iisacc-preview" EXCLUDE PATTERN ".iisacc-temporary.lock" EXCLUDE PATTERN ".DS_Store" EXCLUDE)
install(DIRECTORY tests/fixtures/ DESTINATION ${CMAKE_INSTALL_DATADIR}/iiCSMIDI/tests/fixtures
    PATTERN ".*.iisacc-preview" EXCLUDE PATTERN ".iisacc-temporary.lock" EXCLUDE PATTERN ".DS_Store" EXCLUDE)
