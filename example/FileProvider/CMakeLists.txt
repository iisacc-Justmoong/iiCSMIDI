cmake_minimum_required(VERSION 3.31)
# iisdk:consumer-application
project(iiCSMIDIFileConsumer LANGUAGES CXX)
include(CTest)
find_package(iiCSMIDI 0.3 CONFIG REQUIRED)
find_package(iiFileProviderNative 0.8 CONFIG REQUIRED)
find_package(iiJson 0.1 CONFIG REQUIRED)
add_library(iiCSMIDIFileAdapter INTERFACE)
target_compile_features(iiCSMIDIFileAdapter INTERFACE cxx_std_23)
target_include_directories(iiCSMIDIFileAdapter INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/App")
target_link_libraries(iiCSMIDIFileAdapter INTERFACE iiCSMIDI::iiCSMIDI iiFileProvider::Native iiJson::iiJson)
if(BUILD_TESTING)
    add_executable(iiCSMIDI_authorship tests/authorship.cpp)
    target_link_libraries(iiCSMIDI_authorship PRIVATE iiCSMIDIFileAdapter)
    add_test(NAME iiCSMIDI.authorship COMMAND iiCSMIDI_authorship "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/fixtures")
    set_tests_properties(iiCSMIDI.authorship PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/resources")
endif()
